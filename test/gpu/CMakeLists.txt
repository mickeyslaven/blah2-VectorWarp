cmake_minimum_required(VERSION 3.16)
project(blah2GpuChecks LANGUAGES CXX)
include(CTest)
find_package(Threads REQUIRED)
find_package(Armadillo REQUIRED)
find_library(BLAH2_FFTW3_LIBRARY NAMES fftw3 REQUIRED)
find_library(BLAH2_FFTW3_THREADS_LIBRARY NAMES fftw3_threads REQUIRED)
get_filename_component(PROJECT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
file(MAKE_DIRECTORY "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}")
include_directories(${PROJECT_ROOT}/src)
include(${PROJECT_ROOT}/cmake/RapidJson.cmake)
include(${PROJECT_ROOT}/cmake/GpuAcceleration.cmake)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux" OR APPLE)
  add_executable(testVulkanSmall TestVulkanSmall.cpp)
  target_link_libraries(testVulkanSmall PRIVATE blah2GpuProcess)
  if(TARGET blah2GpuWorker)
    add_dependencies(testVulkanSmall blah2GpuWorker)
  endif()
endif()
foreach(name IN ITEMS Acceleration AccelerationFallback)
  add_executable(test${name} ${PROJECT_ROOT}/test/unit/process/ambiguity/Test${name}.cpp
    ${PROJECT_ROOT}/src/process/ambiguity/Acceleration.cpp
    ${PROJECT_ROOT}/src/process/ambiguity/Ambiguity.cpp
    ${PROJECT_ROOT}/src/process/meta/HammingNumber.cpp
    ${PROJECT_ROOT}/src/data/IqData.cpp ${PROJECT_ROOT}/src/data/Map.cpp)
  target_compile_features(test${name} PRIVATE cxx_std_17)
  target_link_libraries(test${name} PRIVATE blah2RapidJson blah2GpuProcess ${BLAH2_FFTW3_LIBRARY} ${BLAH2_FFTW3_THREADS_LIBRARY} Threads::Threads ${CMAKE_DL_LIBS})
  if(TARGET blah2GpuVulkan)
    add_dependencies(test${name} blah2GpuWorker)
  endif()
endforeach()
add_test(NAME cpuFallback COMMAND testAccelerationFallback)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux" OR APPLE)
  include(${PROJECT_ROOT}/cmake/GpuProcessTests.cmake)
endif()
